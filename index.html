// ============================================================ // APPS SCRIPT — Sistem Absen Piket Deteni // Cara pasang: Ekstensi → Apps Script → paste kode ini // → Deploy → New deployment → Web App // → Execute as: Me | Who has access: Anyone // ============================================================ const SHEET_LAPORAN = 'Laporan Piket'; const SHEET_DETENI = 'Data Deteni'; const SHEET_REKAP = 'Rekap Otomatis'; // ── Terima POST dari web app ────────────────────────────────── function doPost(e) { try { const data = JSON.parse(e.postData.contents); // Abaikan data test if (data.isTest) { return jsonResponse({ status: 'ok', message: 'Test koneksi berhasil' }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const shLap = getOrCreateSheet(ss, SHEET_LAPORAN, headerLaporan()); const shDet = getOrCreateSheet(ss, SHEET_DETENI, headerDeteni()); // Baris laporan utama const rowId = new Date().getTime().toString(); shLap.appendRow([ rowId, data.tanggal, data.jam, data.jamPiket, data.petugas, data.waPetugas, data.statusPetugas, data.kondisi, data.keteranganTA, data.jumlahDeteni, data.fotoKondisi, data.fotoMakan, new Date() // timestamp server ]); // Baris per deteni const deteniList = JSON.parse(data.deteniJson || '[]'); deteniList.forEach(function(d) { shDet.appendRow([ rowId, data.tanggal, data.jam, data.petugas, d.nama, d.jk, d.wn, d.status, d.deportTgl || '-', d.deportJam || '-', data.kondisi ]); }); // Update sheet rekap otomatis updateRekap(ss); return jsonResponse({ status: 'ok', rowId: rowId }); } catch (err) { return jsonResponse({ status: 'error', message: err.toString() }); } } // ── GET: ambil data rekap (opsional, untuk ditampilkan di web) ── function doGet(e) { const action = e.parameter.action || 'rekap'; const ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'rekap') { const period = e.parameter.period || 'minggu'; const shLap = ss.getSheetByName(SHEET_LAPORAN); if (!shLap) return jsonResponse({ rows: [] }); const rows = shLap.getDataRange().getValues(); const now = new Date(); const cutoff = period === 'bulan' ? new Date(now.getFullYear(), now.getMonth(), 1) : new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000); const result = []; for (let i = 1; i < rows.length; i++) { const tglStr = rows[i][1]; // format dd/mm/yyyy const parts = tglStr.split('/'); const rowDate = new Date(parts[2], parts[1]-1, parts[0]); if (rowDate >= cutoff) { result.push({ id: rows[i][0], tanggal: rows[i][1], jam: rows[i][2], jamPiket: rows[i][3], petugas: rows[i][4], kondisi: rows[i][7], jumlahDeteni: rows[i][9] }); } } return jsonResponse({ rows: result }); } return jsonResponse({ status: 'ok' }); } // ── Helper: buat / ambil sheet dengan header ───────────────── function getOrCreateSheet(ss, name, headers) { let sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); formatHeader(sh); } return sh; } function formatHeader(sh) { const range = sh.getRange(1, 1, 1, sh.getLastColumn()); range.setBackground('#0f2d4a') .setFontColor('#ffffff') .setFontWeight('bold') .setFontSize(11); sh.setFrozenRows(1); } function headerLaporan() { return [ 'ID Laporan', 'Tanggal', 'Jam Lapor', 'Jam Piket', 'Nama Petugas', 'WA Petugas', 'Status Petugas', 'Kondisi Keamanan', 'Keterangan TA', 'Jumlah Deteni', 'Foto Kondisi', 'Foto Makan', 'Timestamp Server' ]; } function headerDeteni() { return [ 'ID Laporan', 'Tanggal', 'Jam', 'Petugas', 'Nama Deteni', 'Jenis Kelamin', 'Warga Negara', 'Status Penahanan', 'Tgl Deportasi', 'Jam Deportasi', 'Kondisi Keamanan' ]; } // ── Rekap otomatis: hitung statistik per minggu & bulan ────── function updateRekap(ss) { let sh = ss.getSheetByName(SHEET_REKAP); if (!sh) { sh = ss.insertSheet(SHEET_REKAP); } sh.clearContents(); const shLap = ss.getSheetByName(SHEET_LAPORAN); if (!shLap || shLap.getLastRow() < 2) { sh.getRange(1,1).setValue('Belum ada data laporan.'); return; } const rows = shLap.getDataRange().getValues().slice(1); // skip header const now = new Date(); const periodDefs = [ { label: 'Minggu Ini', cutoff: new Date(now.getTime() - 7*24*60*60*1000) }, { label: 'Bulan Ini', cutoff: new Date(now.getFullYear(), now.getMonth(), 1) }, { label: 'Semua Data', cutoff: new Date(2000, 0, 1) } ]; let output = [ ['REKAP OTOMATIS — Sistem Piket Deteni', '', ''], ['Diperbarui:', Utilities.formatDate(now, 'Asia/Jakarta', 'dd/MM/yyyy HH:mm'), ''], ['', '', ''], ['Periode', 'Total Laporan', 'Jml Deteni', 'Kondisi Aman', 'Kondisi Tidak Aman'] ]; periodDefs.forEach(function(p) { const filtered = rows.filter(function(r) { const parts = String(r[1]).split('/'); if (parts.length < 3) return false; const d = new Date(parts[2], parts[1]-1, parts[0]); return d >= p.cutoff; }); const aman = filtered.filter(function(r){ return r[7] === 'AMAN'; }).length; const tAman = filtered.filter(function(r){ return r[7] === 'TIDAK AMAN'; }).length; const deteni = filtered.reduce(function(sum, r){ return sum + (Number(r[9])||0); }, 0); output.push([p.label, filtered.length, deteni, aman, tAman]); }); output.push(['', '', '', '', '']); output.push(['Detail per Petugas (Semua Data)', '', '', '', '']); output.push(['Nama Petugas', 'Total Laporan', 'Aman', 'Tidak Aman', 'Rata-rata Deteni']); // Group by petugas const byPetugas = {}; rows.forEach(function(r) { const nama = r[4]; if (!byPetugas[nama]) byPetugas[nama] = { total:0, aman:0, tAman:0, deteni:0 }; byPetugas[nama].total++; if (r[7]==='AMAN') byPetugas[nama].aman++; else byPetugas[nama].tAman++; byPetugas[nama].deteni += (Number(r[9])||0); }); Object.keys(byPetugas).forEach(function(nama) { const d = byPetugas[nama]; output.push([nama, d.total, d.aman, d.tAman, (d.deteni / d.total).toFixed(1)]); }); sh.getRange(1, 1, output.length, 5).setValues(output); sh.getRange(1,1,1,5).merge().setBackground('#0f2d4a').setFontColor('#fff').setFontWeight('bold').setFontSize(13); sh.getRange(4,1,1,5).setBackground('#c9a227').setFontWeight('bold'); sh.autoResizeColumns(1,5); } // ── JSON response helper ────────────────────────────────────── function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
